cmake_minimum_required(VERSION 3.16)
project(CppParkingLotManagementSystem VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
add_executable(parking-lot src/main.cpp src/fee_calculator.cpp)
target_include_directories(parking-lot PRIVATE include)
target_link_libraries(parking-lot PRIVATE SQLite3::SQLite3)
if(MSVC)
  target_compile_options(parking-lot PRIVATE /W4 /permissive-)
else()
  target_compile_options(parking-lot PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(parking-tests tests/test_fee_calculator.cpp src/fee_calculator.cpp)
target_include_directories(parking-tests PRIVATE include)
add_test(NAME fee-calculator COMMAND parking-tests)
